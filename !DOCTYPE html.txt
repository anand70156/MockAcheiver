    // State Variables
    let currentQIndex = 0;
    const userAnswers = new Array(examQuestions.length).fill(null);
    const reviewStatus = new Array(examQuestions.length).fill(false);
    let isTestSubmitted = false;
    let totalSecondsLeft = 15 * 60; // 15 Minutes
    let timerInterval = null;
    let timeTakenSeconds = 0;

    // Initialize Web App
    window.onload = function() {
      initTimer();
      renderQuestionPalette();
      loadQuestion(0);
    };

    // 2. Timer System
    function initTimer() {
      const timerDisplay = document.getElementById("countdownTimer");
      timerInterval = setInterval(() => {
        if (totalSecondsLeft <= 0) {
          clearInterval(timerInterval);
          finalizeAndSubmitTest();
          return;
        }

        totalSecondsLeft--;
        timeTakenSeconds++;

        const m = Math.floor(totalSecondsLeft / 60);
        const s = totalSecondsLeft % 60;
        timerDisplay.textContent = `${m < 10 ? '0' : ''}${m}:${s < 10 ? '0' : ''}${s}`;

        // Blink Red on last 2 minutes
        if (totalSecondsLeft <= 120) {
          timerDisplay.classList.add("text-rose-400", "animate-pulse");
        }
      }, 1000);
    }

    // 3. Question Palette Rendering
    function renderQuestionPalette() {
      const palette = document.getElementById("paletteGrid");
      palette.innerHTML = "";

      examQuestions.forEach((q, idx) => {
        const btn = document.createElement("button");
        btn.id = `palette_btn_${idx}`;
        btn.textContent = idx + 1;
        btn.className = getPaletteBtnClass(idx);
        btn.onclick = () => loadQuestion(idx);
        palette.appendChild(btn);
      });
    }

    function getPaletteBtnClass(idx) {
      const baseClass = "h-10 rounded-xl font-bold text-xs flex items-center justify-center transition-all duration-200 ";
      
      if (isTestSubmitted) {
        const isCorrect = userAnswers[idx] === examQuestions[idx].correct;
        const isAttempted = userAnswers[idx] !== null;
        if (!isAttempted) return baseClass + "bg-slate-100 text-slate-400 border border-slate-200";
        if (isCorrect) return baseClass + "bg-emerald-500 text-white shadow-sm shadow-emerald-500/30";
        return baseClass + "bg-rose-500 text-white shadow-sm shadow-rose-500/30";
      }

      if (idx === currentQIndex) {
        return baseClass + "bg-brand-600 text-white ring-4 ring-brand-100 shadow-md";
      }
      if (reviewStatus[idx]) {
        return baseClass + "bg-amber-400 text-slate-900 shadow-sm";
      }
      if (userAnswers[idx] !== null) {
        return baseClass + "bg-emerald-500 text-white shadow-sm";
      }
      return baseClass + "bg-slate-100 text-slate-700 hover:bg-slate-200";
    }

    function refreshPaletteUI() {
      examQuestions.forEach((_, idx) => {
        const btn = document.getElementById(`palette_btn_${idx}`);
        if (btn) btn.className = getPaletteBtnClass(idx);
      });
    }

    // 4. Load & Render Current Question
    function loadQuestion(index) {
      currentQIndex = index;
      const qData = examQuestions[index];

      // Update Header info
      document.getElementById("qNumberDisplay").textContent = index + 1;
      document.getElementById("qProgressText").textContent = `${index + 1} of ${examQuestions.length}`;
      document.getElementById("qSubjectBadge").textContent = qData.subject;
      document.getElementById("qTextDisplay").innerHTML = qData.question;

      // Update Navigation Buttons
      document.getElementById("prevBtn").disabled = index === 0;
      const nextBtn = document.getElementById("nextBtn");
      nextBtn.textContent = index === examQuestions.length - 1 ? "Review All" : "Save & Next →";

      // Render Options
      const optionsBox = document.getElementById("optionsContainer");
      optionsBox.innerHTML = "";
      const letters = ["A", "B", "C", "D"];

      qData.options.forEach((opt, oIdx) => {
        const isSelected = userAnswers[index] === oIdx;
        const optBtn = document.createElement("div");
        optBtn.id = `opt_row_${oIdx}`;

        // Color coding if test is submitted
        let stateStyling = "border-slate-200 hover:border-slate-300 bg-white";
        let badgeStyling = "bg-slate-100 text-slate-700 border border-slate-200";

        if (isTestSubmitted) {
          const isCorrectAnswer = oIdx === qData.correct;
          if (isCorrectAnswer) {
            stateStyling = "border-emerald-500 bg-emerald-50/60 text-emerald-950 font-semibold";
            badgeStyling = "bg-emerald-600 text-white font-bold";
          } else if (isSelected && !isCorrectAnswer) {
            stateStyling = "border-rose-400 bg-rose-50/60 text-rose-950 font-semibold";
            badgeStyling = "bg-rose-600 text-white font-bold";
          }
        } else if (isSelected) {
          stateStyling = "border-brand-600 bg-brand-50/80 text-brand-900 font-semibold shadow-sm ring-2 ring-brand-100";
          badgeStyling = "bg-brand-600 text-white font-bold";
        }

        optBtn.className = `p-3.5 sm:p-4 rounded-2xl border-2 cursor-pointer flex items-center justify-between gap-3 transition-all duration-200 ${stateStyling}`;
        
        optBtn.innerHTML = `
          <div class="flex items-center gap-3.5 flex-1">
            <span class="w-8 h-8 rounded-xl flex items-center justify-center text-xs font-bold shrink-0 transition-colors ${badgeStyling}">
              ${letters[oIdx]}
            </span>
            <span class="text-sm sm:text-base leading-snug">${opt}</span>
          </div>
          ${isTestSubmitted && oIdx === qData.correct ? `
            <span class="text-emerald-600 font-bold text-xs bg-emerald-100 px-2 py-1 rounded-md shrink-0">
              ✓ Correct
            </span>` : ''}
          ${isTestSubmitted && isSelected && oIdx !== qData.correct ? `
            <span class="text-rose-600 font-bold text-xs bg-rose-100 px-2 py-1 rounded-md shrink-0">
              ✕ Your Choice
            </span>` : ''}
        `;

        if (!isTestSubmitted) {
          optBtn.onclick = () => selectOption(index, oIdx);
        }

        optionsBox.appendChild(optBtn);
      });

      // Handle Explanation Box
      const solutionBox = document.getElementById("solutionReviewBox");
      if (isTestSubmitted) {
        solutionBox.classList.remove("hidden");
        const userChoice = userAnswers[index];
        const isCorrect = userChoice === qData.correct;
        const isAttempted = userChoice !== null;

        let statusBanner = "";
        if (!isAttempted) {
          statusBanner = `<div class="text-amber-800 bg-amber-50 p-3 rounded-xl border border-amber-200 font-semibold text-xs flex items-center gap-2">⚠️ आपने यह सवाल हल नहीं किया था (Unattempted)</div>`;
        } else if (isCorrect) {
          statusBanner = `<div class="text-emerald-800 bg-emerald-50 p-3 rounded-xl border border-emerald-200 font-semibold text-xs flex items-center gap-2">✅ बिल्कुल सही उत्तर! (+1 Mark)</div>`;
        } else {
          statusBanner = `<div class="text-rose-800 bg-rose-50 p-3 rounded-xl border border-rose-200 font-semibold text-xs flex items-center gap-2">❌ आपका उत्तर गलत था। सही विकल्प: ${letters[qData.correct]} (${qData.options[qData.correct]})</div>`;
        }

        solutionBox.innerHTML = `
          <div class="p-5 sm:p-6 bg-slate-900 text-white rounded-2xl space-y-4">
            ${statusBanner}
            
            <div>
              <span class="text-[11px] font-bold text-amber-400 tracking-wider uppercase">Comprehensive Concept Breakdown</span>
              <h4 class="text-base font-bold text-white mt-1">${qData.summaryTitle}</h4>
            </div>

            <ul class="space-y-2 text-xs sm:text-sm text-slate-300 leading-relaxed list-disc list-inside">
              ${qData.summaryPoints.map(pt => `<li>${pt}</li>`).join("")}
            </ul>

            <div class="mt-4 p-3 bg-white/5 border border-white/10 rounded-xl text-xs text-indigo-200 flex items-start gap-2">
              <span class="shrink-0 text-sm">📌</span>
              <span><strong>Exam Revision Note:</strong> ${qData.examTip}</span>
            </div>
          </div>
        `;
      } else {
        solutionBox.classList.add("hidden");
      }

      refreshPaletteUI();
    }

    // 5. User Interaction Handlers
    function selectOption(qIdx, oIdx) {
      if (isTestSubmitted) return;
      userAnswers[qIdx] = oIdx;
      // Once answered, remove review mark if set
      reviewStatus[qIdx] = false;
      loadQuestion(qIdx);
    }

    function clearCurrentResponse() {
      if (isTestSubmitted) return;
      userAnswers[currentQIndex] = null;
      reviewStatus[currentQIndex] = false;
      loadQuestion(currentQIndex);
    }

    function toggleReviewCurrent() {
      if (isTestSubmitted) return;
      reviewStatus[currentQIndex] = !reviewStatus[currentQIndex];
      refreshPaletteUI();
    }

    function nextQuestion() {
      if (currentQIndex < examQuestions.length - 1) {
        loadQuestion(currentQIndex + 1);
      } else {
        // If at last question, prompt submit modal
        confirmSubmitModal();
      }
    }

    function prevQuestion() {
      if (currentQIndex > 0) {
        loadQuestion(currentQIndex - 1);
      }
    }

    // 6. Test Submission & Analytics Calculation
    function confirmSubmitModal() {
      let attempted = 0;
      let review = 0;
      userAnswers.forEach((ans, idx) => {
        if (ans !== null) attempted++;
        if (reviewStatus[idx]) review++;
      });

      document.getElementById("modalAttempted").textContent = attempted;
      document.getElementById("modalReview").textContent = review;
      document.getElementById("modalLeft").textContent = examQuestions.length - attempted;
      
      const modal = document.getElementById("submitModal");
      modal.classList.remove("hidden");
      modal.classList.add("flex");
    }

    function closeSubmitModal() {
      const modal = document.getElementById("submitModal");
      modal.classList.add("hidden");
      modal.classList.remove("flex");
    }

    function finalizeAndSubmitTest() {
      closeSubmitModal();
      clearInterval(timerInterval);
      isTestSubmitted = true;

      // Calculate Metrics
      let correct = 0;
      let wrong = 0;
      let skipped = 0;

      examQuestions.forEach((q, idx) => {
        if (userAnswers[idx] === null) {
          skipped++;
        } else if (userAnswers[idx] === q.correct) {
          correct++;
        } else {
          wrong++;
        }
      });

      const total = examQuestions.length;
      const accuracy = (correct + wrong) > 0 ? Math.round((correct / (correct + wrong)) * 100) : 0;
      
      // Update Results DOM
      document.getElementById("resFinalScore").textContent = `${correct} / ${total}`;
      document.getElementById("resAccuracy").textContent = `${accuracy}%`;
      document.getElementById("resCorrectCount").textContent = correct;
      document.getElementById("resWrongCount").textContent = wrong;
      document.getElementById("resSkippedCount").textContent = skipped;

      const mTaken = Math.floor(timeTakenSeconds / 60);
      const sTaken = timeTakenSeconds % 60;
      document.getElementById("timeTakenBadge").textContent = `⏱ Time Taken: ${mTaken}m ${sTaken}s`;

      const rankBadge = document.getElementById("rankBadge");
      if (accuracy >= 80) {
        rankBadge.className = "inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-emerald-500/20 text-emerald-300 text-sm font-semibold border border-emerald-500/30";
        rankBadge.textContent = "🏆 उत्कृष्ट तैयारी (Excellent Performance)";
      } else if (accuracy >= 50) {
        rankBadge.className = "inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-amber-500/20 text-amber-300 text-sm font-semibold border border-amber-500/30";
        rankBadge.textContent = "⚡ अच्छा प्रयास (Needs Revision)";
      } else {
        rankBadge.className = "inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-rose-500/20 text-rose-300 text-sm font-semibold border border-rose-500/30";
        rankBadge.textContent = "📖 अधिक अभ्यास की आवश्यकता (Practice More)";
      }

      // Hide active controls, display result overview
      document.getElementById("resultsSection").classList.remove("hidden");
      document.getElementById("testLiveControls").classList.add("hidden");
      document.getElementById("finishTestBtn").disabled = true;
      document.getElementById("finishTestBtn").classList.add("opacity-50", "cursor-not-allowed");
      document.getElementById("finishTestBtn").textContent = "Test Completed";

      // Refresh current question to reveal solution and color cards
      loadQuestion(0);
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function toggleMobilePalette() {
      const palette = document.querySelector("aside");
      palette.scrollIntoView({ behavior: 'smooth' });
    }
  </script>
